```sql
  (select count(distinct created_by) from public.flashcard_pages)             as so_nguoi_soan,
  (select max(created_at)::date from public.flashcard_pages)                  as soan_gan_nhat;

-- (2) Liệt kê để NHÌN xem có bộ thẻ nào do người thật soạn không
select d.title as bo_the, s.session_number as buoi, s.title as ten_buoi,
       count(p.id) filter (where p.kind = 'vocabulary') as so_the_tu_vung,
       max(p.created_at)::date as soan_gan_nhat
from public.flashcard_decks d
left join public.flashcard_sections s on s.deck_id = d.id
left join public.flashcard_pages   p on p.section_id = s.id
group by d.title, s.session_number, s.title
order by d.title, s.session_number;
```
